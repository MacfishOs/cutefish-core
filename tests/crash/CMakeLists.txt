cmake_minimum_required(VERSION 3.22)
project(coredump_test LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Core Test)
set(CRASH_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../../cutefish-services/crash)
add_executable(coredump_test coredump_test.cpp ${CRASH_DIR}/coredump.cpp)
target_include_directories(coredump_test PRIVATE ${CRASH_DIR})
target_link_libraries(coredump_test PRIVATE Qt6::Core Qt6::Test)
enable_testing()
add_test(NAME coredump_test COMMAND coredump_test)
